    88→/* ── Nested tree ── */
    89→
    90→interface TreeNode {
    91→  name: string;
    92→  path: string;
    93→  type: "file" | "dir";
    94→  size: number;
    95→  children: TreeNode[];
    96→}
    97→
    98→function buildTree(entries: FileEntry[]): TreeNode[] {
    99→  const root: TreeNode = { name: "", path: "", type: "dir", size: 0, children: [] };
   100→  for (const entry of entries) {
   101→    const segments = entry.path.split("/");
   102→    let node = root;
   103→    segments.forEach((seg, i) => {
   104→      const isLeaf = i === segments.length - 1;
   105→      const path = segments.slice(0, i + 1).join("/");
   106→      let child = node.children.find((c) => c.name === seg);
   107→      if (!child) {
   108→        child = {
   109→          name: seg,
   110→          path,
   111→          type: isLeaf ? entry.type : "dir",
   112→          size: isLeaf ? entry.size : 0,
   113→          children: [],
   114→        };
   115→        node.children.push(child);
   116→      }
   117→      node = child;
   118→    });
   119→  }
   120→  const sortRec = (n: TreeNode) => {
   121→    n.children.sort((a, b) => {
   122→      if (a.type !== b.type) return a.type === "dir" ? -1 : 1;
   123→      return a.name.localeCompare(b.name, "ru");
   124→    });
   125→    n.children.forEach(sortRec);
   126→  };
   127→  sortRec(root);
   128→  return root.children;
   129→}

   170→

   177→      const loaded = await api.getProject(projectId);

   180→      if (seq === projectSeqRef.current) {
   181→        toast.error(err instanceof ApiError ? err.message : "Не удалось загрузить проект");
   182→        closeProject();
   183→      }
   184→    } finally {
   185→      if (seq === projectSeqRef.current) setProjectLoading(false);
   186→    }
   187→  }, [projectId, closeProject]);
   188→
   189→  const loadTree = useCallback(
   190→    async (silent: boolean) => {

   192→      if (!silent) setTreeLoading(true);
   193→      try {
   194→        const res = await api.getProjectTree(projectId);
   195→        if (seq !== treeSeqRef.current) return;
   196→        setTree(res.tree);
   197→        setTreeTruncated(res.truncated);
   198→      } catch (err) {
   199→        if (seq === treeSeqRef.current) {
   200→          toast.error(err instanceof ApiError ? err.message : "Не удалось загрузить файлы");
   201→        }
   202→      } finally {
   203→        if (seq === treeSeqRef.current) setTreeLoading(false);
   204→      }
   205→    },
   206→    [projectId],
   207→  );
   208→
   209→  useEffect(() => {
   210→    setProject(null);
   211→    setTree([]);
   212→    setOpenFiles([]);
   213→    setActivePath(null);

   215→    void loadTree(false);
   216→  }, [loadProject, loadTree]);
   217→
   218→  // WS / store version bump → refresh tree + stats (+ active file when clean).

   403→        <div className="flex min-w-0 flex-1 items-center gap-2">

   932→function CheckpointDialog({
   933→  open,
   934→  onOpenChange,
   935→  projectId,
   936→  onDone,
   937→}: {
   938→  open: boolean;
   939→  onOpenChange: (open: boolean) => void;
   940→  projectId: string;
   941→  onDone: () => void;
   942→}) {
   943→  const [message, setMessage] = useState("Изменения из редактора");
   944→
   945→  const [submitting, setSubmitting] = useState(false);
   946→  const [error, setError] = useState<string | null>(null);

   967→      setSubmitting(false);
   968→    }

   972→    <Dialog open={open} onOpenChange={onOpenChange}>
   973→      <DialogContent className="top-[50%] max-h-[85vh] translate-y-[-50%] overflow-y-auto gap-0 sm:max-w-md" aria-describedby={undefined}>
   974→        <DialogHeader className="pb-3">
   975→          <DialogTitle className="flex items-center gap-2 text-sm">

   986→              Сообщение коммита
   987→            </label>
   988→            <Input
   989→              id="checkpoint-message"
   990→              value={message}
   991→              onChange={(e) => setMessage(e.target.value)}
   992→              maxLength={200}
   993→              disabled={submitting}
   994→              autoFocus
   995→            />
   996→          </div>

   998→            <p className="text-xs text-destructive" role="alert">
   999→              {error}
  1000→            </p>
  1001→          )}
  1002→          <Button
  1003→            className="h-10 w-full gap-2 rounded-xl"
  1004→            onClick={() => void submit()}
  1005→            disabled={submitting || message.trim().length === 0}

  1009→                <Loader2 className="size-4 animate-spin" aria-hidden="true" />
