     3→/**
     4→ * ThreadsProvider — thread list + active thread messages, REST actions and
     5→ * live WS subscriptions (agent-service :3003).
     6→ *
     7→ * Flow: optimistic user message → "message:user" replaces it →
     8→ * "agent:thinking" → "message:start" adds an empty assistant bubble →
     9→ * "message:delta"× appends → "message:end" finalizes. "thread:updated"
    10→ * renames in the list (auto-title), "error" → toast + busy-state cleanup.
    11→ */

    50→interface ThreadsContextValue {
    51→  threads: ThreadListItem[];
    52→  threadsLoading: boolean;
    53→  activeThreadId: string | null;
    54→  activeThread: ThreadListItem | null;
    55→  messages: ChatMessage[];
    56→  messagesLoading: boolean;
    57→  /** Agent is thinking or streaming in the active thread — composer disabled. */
    58→  busy: boolean;
    59→  /** "agent:thinking" received for the active thread (before the bubble appears). */
    60→  thinking: boolean;
    61→  /** Orchestrator phase for the active thread (plan/act/review) + label. */
    62→  phase: { phase: TurnPhase; label: string | null } | null;
    63→  /** Plan checklist of the active thread (live via "tasks:updated"). */
    64→  tasks: Task[];
    65→  selectThread: (id: string) => Promise<void>;
    66→  newThread: () => Promise<void>;
    67→  deleteThread: (id: string) => Promise<void>;
    68→  renameThread: (id: string, title: string) => Promise<void>;
    69→  /** Switch the thread mode (ask/plan/act/review) — optimistic + PATCH. */
    70→  updateThreadMode: (id: string, mode: ThreadMode) => Promise<void>;
    71→  /**
    72→   * Create a new thread bound to a project («Обсудить проект») and make it
    73→   * active. Returns the thread or null on failure.
    74→   */
    75→  startProjectThread: (projectId: string, title: string) => Promise<void>;
    76→  sendMessage: (content: string) => Promise<void>;
    77→}
    78→
    79→const ThreadsContext = createContext<ThreadsContextValue | null>(null);
    80→
    81→function tempId(): string {
    82→  return `temp-${Date.now()}-${Math.random().toString(36).slice(2, 8)}`;
    83→}
    84→
    85→/**
    86→ * Normalize an agent tool result (create_note / open_note) into a Note for
    87→ * the context panel. create_note nests the note under result.note with a
    88→ * sibling result.category; open_note returns the note itself.
    89→ */
    90→function noteFromToolResult(tool: string, result: unknown): Note | null {
    91→  if (typeof result !== "object" || result === null) return null;
    92→  const r = result as Record<string, unknown>;
    93→  const raw = tool === "create_note" ? r.note : r;
    94→  if (typeof raw !== "object" || raw === null) return null;
    95→  const n = raw as Record<string, unknown>;
    96→  if (typeof n.id !== "string") return null;
    97→
    98→  const cat = (tool === "create_note" ? r.category : n.category) as
    99→    | Record<string, unknown>
   100→    | null
   101→    | undefined;
   102→
   103→  return {
   104→    id: n.id,
   105→    rawText: typeof n.rawText === "string" ? n.rawText : null,
   106→    status: (typeof n.status === "string" ? n.status : "pending") as NoteStatus,
   107→    favorite: typeof n.favorite === "boolean" ? n.favorite : false,
   108→    createdAt:
   109→      typeof n.createdAt === "string"
   110→        ? n.createdAt
   111→        : new Date().toISOString(),
   112→    category:
   113→      cat && typeof cat === "object" && typeof cat.id === "string"
   114→        ? {
   115→            id: String(cat.id),
   116→            name: String(cat.name ?? ""),
   117→            color: String(cat.color ?? "stone"),
   118→            icon: String(cat.icon ?? "lightbulb"),
   119→          }
   120→        : null,
   121→  };
   122→}
   123→
   124→export function ThreadsProvider({ children }: { children: ReactNode }) {
   125→  const { socket, ensureConnected } = useSocket();
   126→
   127→  const [threads, setThreads] = useState<ThreadListItem[]>([]);
   128→  const [threadsLoading, setThreadsLoading] = useState(true);
   129→  const [activeThreadId, setActiveThreadId] = useState<string | null>(null);
   130→  const [messages, setMessages] = useState<ChatMessage[]>([]);
   131→  const [messagesLoading, setMessagesLoading] = useState(false);
   132→  const [thinkingThreadId, setThinkingThreadId] = useState<string | null>(null);
   133→  const [tasks, setTasks] = useState<Task[]>([]);
   134→  const [phase, setPhase] = useState<{
   135→    threadId: string;
   136→    phase: TurnPhase;
   137→    label: string | null;
   138→  } | null>(null);
   139→  const [streaming, setStreaming] = useState<{
   140→    threadId: string;
   141→    messageId: string;
   142→  } | null>(null);
   143→
   144→  // Refs mirror state so WS handlers and callbacks always see fresh values.
   145→  const socketRef = useRef(socket);
   146→  const activeIdRef = useRef<string | null>(null);
   147→  const messagesRef = useRef<ChatMessage[]>([]);
   148→  const threadsRef = useRef<ThreadListItem[]>([]);
   149→  const streamingRef = useRef<typeof streaming>(null);
   150→  /** Thread whose messages finished loading — required for safe empty-thread cleanup. */
   151→  const loadedRef = useRef<string | null>(null);
   152→  const selectSeqRef = useRef(0);
   153→
   154→  useEffect(() => {
   155→    socketRef.current = socket;
   156→  }, [socket]);
   157→  useEffect(() => {
   158→    activeIdRef.current = activeThreadId;
   159→  }, [activeThreadId]);
   160→  useEffect(() => {
   161→    messagesRef.current = messages;
   162→  }, [messages]);
   163→  useEffect(() => {
   164→    threadsRef.current = threads;
   165→  }, [threads]);
   166→  useEffect(() => {
   167→    streamingRef.current = streaming;
   168→  }, [streaming]);
   169→
   170→  /** Update list preview and move the thread to the top (most recently active). */
   171→  const bumpThread = useCallback((threadId: string, message: Message) => {
   172→    setThreads((prev) => {
   173→      const idx = prev.findIndex((t) => t.id === threadId);
   174→      if (idx === -1) return prev;
   175→      const copy = [...prev];
   176→      const item: ThreadListItem = {
   177→        ...copy[idx],
   178→        updatedAt: message.createdAt,
   179→        lastMessage: {
   180→          content: message.content,
   181→          role: message.role,
   182→          createdAt: message.createdAt,
   183→        },
   184→      };
   185→      copy.splice(idx, 1);
   186→      copy.unshift(item);
   187→      return copy;
   188→    });
   189→  }, []);
   190→
   191→  /** Silently delete an empty thread when the user switches away from it. */
   192→  const maybeDeleteEmptyThread = useCallback(async (threadId: string) => {
   193→    // Only delete when messages were fully loaded and the thread is truly empty
   194→    // (no optimistic or streaming messages either).
   195→    if (loadedRef.current !== threadId) return;
   196→    if (messagesRef.current.length > 0) return;
   197→    if (streamingRef.current?.threadId === threadId) return;
   198→    try {
   199→      await api.deleteThread(threadId);
   200→      setThreads((prev) => prev.filter((t) => t.id !== threadId));
   201→    } catch {
   202→      // Keep the thread if the request fails — harmless.
   203→    }
   204→  }, []);
   205→
   206→  const emitLeave = useCallback((threadId: string) => {
   207→    const s = socketRef.current;
   208→    if (s && s.connected) s.emit("thread:leave", { threadId });
   209→  }, []);
   210→
   211→  const selectThreadInternal = useCallback(
   212→    async (id: string, opts?: { skipCleanup?: boolean }) => {
   213→      const prevId = activeIdRef.current;
   214→      if (prevId === id) return;
   215→
   216→      if (prevId && !opts?.skipCleanup) {
   217→        await maybeDeleteEmptyThread(prevId);
   218→      }
   219→      if (prevId) emitLeave(prevId);
   220→
   221→      const seq = ++selectSeqRef.current;
   222→      loadedRef.current = null;
   223→      activeIdRef.current = id;
   224→      setActiveThreadId(id);
   225→      setMessages([]);
   226→      setTasks([]);
   227→      setPhase(null);
   228→      setMessagesLoading(true);
   229→
   230→      try {
   231→        const [{ thread, messages: loaded }, tasksRes] = await Promise.all([
   232→          api.getThread(id),
   233→          api.getThreadTasks(id).catch(() => ({ tasks: [] as Task[] })),
   234→        ]);
   235→        if (seq !== selectSeqRef.current) return; // stale response
   236→        setThreads((prev) => {
   237→          const idx = prev.findIndex((t) => t.id === id);
   238→          if (idx === -1) return prev;
   239→          const copy = [...prev];
   240→          copy[idx] = { ...copy[idx], ...thread };
   241→          return copy;
   242→        });
   243→        loadedRef.current = id;
   244→        setMessages(loaded.map((m) => ({ ...m })));
   245→        setTasks(tasksRes.tasks);
   246→        const s = socketRef.current;
   247→        if (s && s.connected) s.emit("thread:join", { threadId: id });
   248→      } catch {
   249→        if (seq === selectSeqRef.current) {
   250→          toast.error("Не удалось загрузить диалог");
   251→        }
   252→      } finally {
   253→        if (seq === selectSeqRef.current) {
   254→          setMessagesLoading(false);
   255→        }
   256→      }
   257→    },
   258→    [emitLeave, maybeDeleteEmptyThread],
   259→  );
   260→
   261→  // Keep a stable ref to the latest selector for the one-shot initial load.
   262→  const selectRef = useRef(selectThreadInternal);
   263→  useEffect(() => {
   264→    selectRef.current = selectThreadInternal;
   265→  });
   266→
   267→  // Initial load: thread list + auto-select the most recent thread.
   268→  useEffect(() => {
   269→    let cancelled = false;
   270→    (async () => {
   271→      try {
   272→        const list = await api.listThreads();
   273→        if (cancelled) return;
   274→        setThreads(list);
   275→        if (list.length > 0) {
   276→          await selectRef.current(list[0].id, { skipCleanup: true });
   277→        }
   278→      } catch {
   279→        if (!cancelled) toast.error("Не удалось загрузить список диалогов");
   280→      } finally {
   281→        if (!cancelled) setThreadsLoading(false);
   282→      }
   283→    })();
   284→    return () => {
   285→      cancelled = true;
   286→    };
   287→  }, []);
   288→
   289→  const selectThread = useCallback(
   290→    async (id: string) => {
   291→      await selectThreadInternal(id);
   292→    },
   293→    [selectThreadInternal],
   294→  );
   295→
   296→  const newThread = useCallback(async () => {
   297→    const prevId = activeIdRef.current;
   298→    if (prevId) {
   299→      await maybeDeleteEmptyThread(prevId);
   300→      emitLeave(prevId);
   301→    }
   302→    try {
   303→      const thread = await api.createThread();
   304→      const seq = ++selectSeqRef.current;
   305→      setThreads((prev) => [{ ...thread, lastMessage: null }, ...prev]);
   306→      loadedRef.current = thread.id;
   307→      activeIdRef.current = thread.id;
   308→      setActiveThreadId(thread.id);
   309→      setMessages([]);
   310→      setTasks([]);
   311→      setPhase(null);
   312→      setMessagesLoading(false);
   313→      const s = socketRef.current;
   314→      if (s && s.connected) s.emit("thread:join", { threadId: thread.id });
   315→    } catch {
   316→      toast.error("Не удалось создать диалог");
   317→    }
   318→  }, [emitLeave, maybeDeleteEmptyThread]);
   319→
   320→  /** New thread pre-bound to a project (project screen «Обсудить проект»). */
   321→  const startProjectThread = useCallback(
   322→    async (projectId: string, title: string) => {
   323→      const prevId = activeIdRef.current;
   324→      if (prevId) {
   325→        await maybeDeleteEmptyThread(prevId);
   326→        emitLeave(prevId);
   327→      }
   328→      try {
   329→        const thread = await api.createThread({ projectId, title });
   330→        const seq = ++selectSeqRef.current;
   331→        setThreads((prev) => [{ ...thread, lastMessage: null }, ...prev]);
   332→        loadedRef.current = thread.id;
   333→        activeIdRef.current = thread.id;
   334→        setActiveThreadId(thread.id);
   335→        setMessages([]);
   336→        setTasks([]);
   337→        setPhase(null);
   338→        setMessagesLoading(false);
   339→        const s = socketRef.current;
   340→        if (s && s.connected) s.emit("thread:join", { threadId: thread.id });
   341→      } catch {
   342→        toast.error("Не удалось создать диалог с проектом");
   343→      }
   344→    },
   345→    [emitLeave, maybeDeleteEmptyThread],
   346→  );
   347→
   348→  /** Optimistic mode switch; reverts the chip when the PATCH fails. */
   349→  const updateThreadMode = useCallback(
   350→    async (id: string, mode: ThreadMode) => {
   351→      const prevMode = threadsRef.current.find((t) => t.id === id)?.mode;
   352→      setThreads((prev) =>
   353→        prev.map((t) => (t.id === id ? { ...t, mode } : t)),
   354→      );
   355→      try {
   356→        await api.updateThread(id, { mode });
   357→      } catch {
   358→        setThreads((prev) =>
   359→          prev.map((t) =>
   360→            t.id === id && prevMode ? { ...t, mode: prevMode } : t,
   361→          ),
   362→        );
   363→        toast.error("Не удалось изменить режим диалога");
   364→      }
   365→    },
   366→    [],
   367→  );
   368→
   369→  const deleteThread = useCallback(
   370→    async (id: string) => {
   371→      try {
   372→        await api.deleteThread(id);
   373→      } catch {
   374→        toast.error("Не удалось удалить диалог");
   375→        return;
   376→      }
   377→      setThreads((prev) => prev.filter((t) => t.id !== id));
   378→      if (activeIdRef.current === id) {
   379→        emitLeave(id);
   380→        loadedRef.current = null;
   381→        const next = threadsRef.current.find((t) => t.id !== id);
   382→        if (next) {
   383→          await selectThreadInternal(next.id, { skipCleanup: true });
   384→        } else {
   385→          const seq = ++selectSeqRef.current;
   386→          activeIdRef.current = null;
   387→          setActiveThreadId(null);
   388→          setMessages([]);
   389→          setTasks([]);
   390→          setPhase(null);
   391→          setMessagesLoading(false);
   392→        }
   393→      }
   394→    },
   395→    [emitLeave, selectThreadInternal],
   396→  );
   397→
   398→  const renameThread = useCallback(async (id: string, title: string) => {
   399→    const trimmed = title.trim();
   400→    if (!trimmed) return;
   401→    try {
   402→      const thread = await api.updateThread(id, { title: trimmed });
   403→      setThreads((prev) =>
   404→        prev.map((t) => (t.id === id ? { ...t, title: thread.title } : t)),
   405→      );
   406→    } catch {
   407→      toast.error("Не удалось переименовать диалог");
   408→    }
   409→  }, []);
   410→
   411→  const sendMessage = useCallback(
   412→    async (content: string) => {
   413→      const trimmed = content.trim();
   414→      if (!trimmed) return;
   415→
   416→      // No active thread yet (fresh account / all deleted) — create one
   417→      // transparently so the first message always works (Cursor-style).
   418→      let threadId = activeIdRef.current;
   419→      if (!threadId) {
   420→        try {
   421→          const thread = await api.createThread();
   422→          setThreads((prev) => [{ ...thread, lastMessage: null }, ...prev]);
   423→          loadedRef.current = thread.id;
   424→          activeIdRef.current = thread.id;
   425→          setActiveThreadId(thread.id);
   426→          setMessages([]);
   427→          setMessagesLoading(false);
   428→          const s0 = socketRef.current;
   429→          if (s0 && s0.connected) s0.emit("thread:join", { threadId: thread.id });
   430→          threadId = thread.id;
   431→        } catch {
   432→          toast.error("Не удалось создать диалог");
   433→          return;
   434→        }
   435→      }
   436→
   437→      const id = tempId();
   438→      const now = new Date().toISOString();
   439→      setMessages((prev) => [
   440→        ...prev,
   441→        {
   442→          id,
   443→          threadId,
   444→          role: "user",
   445→          content: trimmed,
   446→          createdAt: now,
   447→          pending: true,
   448→        },
   449→      ]);
   450→      bumpThread(threadId, {
   451→        id,
   452→        threadId,
   453→        role: "user",
   454→        content: trimmed,
   455→        createdAt: now,
   456→      });
   457→
   458→      const ok = await ensureConnected();
   459→      const s = socketRef.current;
   460→      if (!ok || !s) {
   461→        toast.error("Нет соединения — сообщение не отправлено");
   462→        setMessages((prev) => prev.filter((m) => m.id !== id));
   463→        return;
   464→      }
   465→      // Make sure we are in the room before sending (idempotent on server).
   466→      s.emit("thread:join", { threadId });
   467→      s.emit("message:send", { threadId, content: trimmed });
   468→    },
   469→    [bumpThread, ensureConnected],
   470→  );
   471→
   472→  // Re-join the active thread room after every (re)connect.
   473→  useEffect(() => {
   474→    if (!socket) return;
   475→    const handleConnect = () => {
   476→      const id = activeIdRef.current;
   477→      if (id) socket.emit("thread:join", { threadId: id });
   478→    };
   479→    socket.on("connect", handleConnect);
   480→    if (socket.connected) handleConnect();
   481→    return () => {
   482→      socket.off("connect", handleConnect);
   483→    };
   484→  }, [socket]);
   485→
   486→  // WS event subscriptions.
   487→  useEffect(() => {
   488→    if (!socket) return;
   489→
   490→    const onMessageUser = ({ message }: WsMessageUserPayload) => {
   491→      if (message.threadId === activeIdRef.current) {
   492→        setMessages((prev) => {
   493→          // Replace the matching optimistic message…
   494→          const reverseIdx = [...prev]
   495→            .reverse()
   496→            .findIndex(
   497→              (m) => m.id.startsWith("temp-") && m.content === message.content,
   498→            );
   499→          if (reverseIdx >= 0) {
   500→            const idx = prev.length - 1 - reverseIdx;
   501→            const copy = [...prev];
   502→            copy[idx] = { ...message };
   503→            return copy;
   504→          }
   505→          // …or append if this came from another tab (dedup by id).
   506→          if (prev.some((m) => m.id === message.id)) return prev;
   507→          return [...prev, { ...message }];
   508→        });
   509→      }
   510→      bumpThread(message.threadId, message);
   511→    };
   512→
   513→    const onAgentThinking = ({ threadId }: WsAgentThinkingPayload) => {
   514→      setThinkingThreadId(threadId);
   515→    };
   516→
   517→    const onTasksUpdated = ({ threadId, tasks: updated }: WsTasksUpdatedPayload) => {
   518→      if (threadId === activeIdRef.current) setTasks(updated);
   519→    };
   520→
   521→    const onTurnPhase = ({ threadId, phase: p, label }: WsTurnPhasePayload) => {
   522→      if (threadId !== activeIdRef.current) return;
   523→      if (p === "idle") {
   524→        setPhase(null);
   525→        return;
   526→      }
   527→      setPhase({ threadId, phase: p, label });
   528→      // A phase event implies the agent is working (keep the indicator up).
   529→      setThinkingThreadId(threadId);
   530→    };
   531→
   532→    const onToolStart = ({
   533→      threadId,
   534→      messageId,
   535→      tool,
   536→      args,
   537→    }: WsToolStartPayload) => {
   538→      // Tool events for background threads are ignored — they will be loaded
   539→      // from the REST history when the thread becomes active.
   540→      if (threadId !== activeIdRef.current) return;
   541→      // The tool card spinner replaces the typing indicator.
   542→      setThinkingThreadId((cur) => (cur === threadId ? null : cur));
   543→      setMessages((prev) => {
   544→        if (prev.some((m) => m.id === messageId)) return prev;
   545→        return [
   546→          ...prev,
   547→          {
   548→            id: messageId,
   549→            threadId,
   550→            role: "tool" as const,
   551→            content: "",
   552→            toolName: tool,
   553→            toolArgs: JSON.stringify(args ?? {}),
   554→            toolResult: null,
   555→            createdAt: new Date().toISOString(),
   556→            toolPending: true,
   557→          },
   558→        ];
   559→      });
   560→    };
   561→
   562→    const onToolEnd = ({
   563→      threadId,
   564→      messageId,
   565→      tool,
   566→      result,
   567→    }: WsToolEndPayload) => {
   568→      if (threadId !== activeIdRef.current) return;
   569→      setMessages((prev) =>
   570→        prev.map((m) =>
   571→          m.id === messageId
   572→            ? {
   573→                ...m,
   574→                toolResult: JSON.stringify(result ?? {}),
   575→                toolPending: false,
   576→              }
   577→            : m,
   578→        ),
   579→      );
   580→
   581→      // create_note / open_note with a note payload → open it in the context
   582→      // panel (auto: never pops the mobile dialog). New notes refresh the
   583→      // notebook feed silently.
   584→      if (tool === "create_note" || tool === "open_note") {
   585→        const note = noteFromToolResult(tool, result);
   586→        if (note) {
   587→          const ui = useAppUi.getState();
   588→          ui.openNote(note, { auto: true });
   589→          if (tool === "create_note") ui.bumpNotes();
   590→        }
   591→      }
   592→    };
   593→
   594→    const onMessageStart = ({ threadId, messageId }: WsMessageStartPayload) => {
   595→      setThinkingThreadId((cur) => (cur === threadId ? null : cur));
   596→      setStreaming({ threadId, messageId });
   597→      if (threadId === activeIdRef.current) {
   598→        setMessages((prev) => [
   599→          ...prev,
   600→          {
   601→            id: messageId,
   602→            threadId,
   603→            role: "assistant",
   604→            content: "",
   605→            createdAt: new Date().toISOString(),
   606→            streaming: true,
   607→          },
   608→        ]);
   609→      }
   610→    };
   611→
   612→    const onMessageDelta = ({
   613→      threadId,
   614→      messageId,
   615→      delta,
   616→    }: WsMessageDeltaPayload) => {
   617→      if (threadId !== activeIdRef.current) return;
   618→      setMessages((prev) => {
   619→        const idx = prev.findIndex((m) => m.id === messageId);
   620→        if (idx >= 0) {
   621→          const copy = [...prev];
   622→          copy[idx] = { ...copy[idx], content: copy[idx].content + delta };
   623→          return copy;
   624→        }
   625→        // User switched back mid-stream: create the bubble on first delta.
   626→        return [
   627→          ...prev,
   628→          {
   629→            id: messageId,
   630→            threadId,
   631→            role: "assistant",
   632→            content: delta,
   633→            createdAt: new Date().toISOString(),
   634→            streaming: true,
   635→          },
   636→        ];
   637→      });
   638→    };
   639→
   640→    const onMessageEnd = ({ threadId, message }: WsMessageEndPayload) => {
   641→      setStreaming((cur) =>
   642→        cur && cur.threadId === threadId ? null : cur,
   643→      );
   644→      setThinkingThreadId((cur) => (cur === threadId ? null : cur));
   645→      setPhase((cur) => (cur && cur.threadId === threadId ? null : cur));
   646→      if (threadId === activeIdRef.current) {
   647→        setMessages((prev) => {
   648→          const idx = prev.findIndex((m) => m.id === message.id);
   649→          if (idx >= 0) {
   650→            const copy = [...prev];
   651→            copy[idx] = { ...message };
   652→            return copy;
   653→          }
   654→          return [...prev, { ...message }];
   655→        });
   656→      }
   657→      bumpThread(threadId, message);
   658→    };
   659→
   660→    const onThreadUpdated = ({ thread }: WsThreadUpdatedPayload) => {
   661→      setThreads((prev) =>
   662→        prev.map((t) =>
   663→          t.id === thread.id
   664→            ? { ...t, title: thread.title, updatedAt: thread.updatedAt }
   665→            : t,
   666→        ),
   667→      );
   668→    };
   669→
   670→    const onError = ({ message }: WsErrorPayload) => {
   671→      toast.error(message);
   672→      setThinkingThreadId(null);
   673→      setPhase(null);
   674→      // A failed turn can leave tool cards stuck in the running state —
   675→      // finalize them without a result.
   676→      setMessages((prev) =>
   677→        prev.some((m) => m.toolPending)
   678→          ? prev.map((m) => (m.toolPending ? { ...m, toolPending: false } : m))
   679→          : prev,
   680→      );
   681→      // If the turn failed before any content arrived, drop the empty
   682→      // assistant bubble and clear the busy flag so the composer unlocks.
   683→      const st = streamingRef.current;
   684→      if (st) {
   685→        const live = messagesRef.current.find((m) => m.id === st.messageId);
   686→        if (!live || live.content === "") {
   687→          setStreaming((cur) =>
   688→            cur && cur.messageId === st.messageId ? null : cur,
   689→          );
   690→          setMessages((prev) =>
   691→            prev.filter(
   692→              (m) => !(m.role === "assistant" && m.streaming && m.content === ""),
   693→            ),
   694→          );
   695→        }
   696→      }
   697→    };
   698→
   699→    socket.on("message:user", onMessageUser);
   700→    socket.on("agent:thinking", onAgentThinking);
   701→    socket.on("message:start", onMessageStart);
   702→    socket.on("message:delta", onMessageDelta);
   703→    socket.on("message:end", onMessageEnd);
   704→    socket.on("thread:updated", onThreadUpdated);
   705→    socket.on("tool:start", onToolStart);
   706→    socket.on("tool:end", onToolEnd);
   707→    socket.on("tasks:updated", onTasksUpdated);
   708→    socket.on("turn:phase", onTurnPhase);
   709→    socket.on("error", onError);
   710→
   711→    return () => {
   712→      socket.off("message:user", onMessageUser);
   713→      socket.off("agent:thinking", onAgentThinking);
   714→      socket.off("message:start", onMessageStart);
   715→      socket.off("message:delta", onMessageDelta);
   716→      socket.off("message:end", onMessageEnd);
   717→      socket.off("thread:updated", onThreadUpdated);
   718→      socket.off("tool:start", onToolStart);
   719→      socket.off("tool:end", onToolEnd);
   720→      socket.off("tasks:updated", onTasksUpdated);
   721→      socket.off("turn:phase", onTurnPhase);
   722→      socket.off("error", onError);
   723→    };
   724→  }, [socket, bumpThread]);
   725→
   726→  const activeThread = useMemo(
   727→    () => threads.find((t) => t.id === activeThreadId) ?? null,
   728→    [threads, activeThreadId],
   729→  );
   730→
   731→  const thinking = thinkingThreadId !== null && thinkingThreadId === activeThreadId;
   732→
   733→  // Tool cards in the running state also mean the agent is still working.
   734→  const toolBusy = messages.some(
   735→    (m) => m.role === "tool" && m.toolPending === true,
   736→  );
   737→
   738→  const activePhase =
   739→    phase && phase.threadId === activeThreadId
   740→      ? { phase: phase.phase, label: phase.label }
   741→      : null;
   742→
   743→  const value = useMemo<ThreadsContextValue>(
   744→    () => ({
   745→      threads,
   746→      threadsLoading,
   747→      activeThreadId,
   748→      activeThread,
   749→      messages,
   750→      messagesLoading,
   751→      busy:
   752→        thinking || streaming?.threadId === activeThreadId || toolBusy,
   753→      thinking,
   754→      phase: activePhase,
   755→      tasks,
   756→      selectThread,
   757→      newThread,
   758→      deleteThread,
   759→      renameThread,
   760→      updateThreadMode,
   761→      startProjectThread,
   762→      sendMessage,
   763→    }),
   764→    [
   765→      threads,
   766→      threadsLoading,
   767→      activeThreadId,
   768→      activeThread,
   769→      messages,
   770→      messagesLoading,
   771→      thinking,
   772→      streaming,
   773→      toolBusy,
   774→      activePhase,
   775→      tasks,
   776→      selectThread,
   777→      newThread,
   778→      deleteThread,
   779→      renameThread,
   780→      updateThreadMode,
   781→      startProjectThread,
   782→      sendMessage,
   783→    ],
   784→  );
   785→
   786→  return (
   787→    <ThreadsContext.Provider value={value}>{children}</ThreadsContext.Provider>
   788→  );
   789→}
   790→
   791→export function useThreads(): ThreadsContextValue {
   792→  const ctx = useContext(ThreadsContext);
   793→  if (!ctx) {
   794→    throw new Error("useThreads must be used within <ThreadsProvider>");
   795→  }
   796→  return ctx;
   797→}
