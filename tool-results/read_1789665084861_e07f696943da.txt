     1→// agent-service — socket.io transport + agent turn orchestration
     2→// (tool-calling loop over the notebook tools + Stage-3 project/file tools).
     3→//
     4→// Contract (worklog Tasks 1, 4 & 3-ctr):
     5→//   Client→server: "thread:join" {threadId}; "thread:leave" {threadId};
     6→//                  "message:send" {threadId, content}
     7→//   Server→client: "message:user" {message}; "agent:thinking" {threadId};
     8→//                  "message:start" {threadId, messageId};
     9→//                  "message:delta" {threadId, messageId, delta};
    10→//                  "message:end" {threadId, message};
    11→//                  "tool:start" {threadId, messageId, tool, args};
    12→//                  "tool:end" {threadId, messageId, tool, args, result};
    13→//                  "thread:updated" {thread}; "error" {message};
    14→//                  "project:created" {project: {id, name, origin}} (user room);
    15→//                  "project:updated" {projectId, reason: "files"|"checkpoint"} (user room)
    16→//   Message shape: {id, threadId, role: 'user'|'assistant', content, createdAt}
    17→//
    18→// Agent turn = up to 8 LLM iterations. Each LLM reply is either plain text
    19→// (→ streamed to the client via message:start/delta/end, loop ends) or a JSON
    20→// tool call (→ tool row persisted, tool executed with ToolContext
    21→// {threadId, mode, projectId} from the thread row, tool:start/tool:end
    22→// emitted, result fed back into the next LLM call via [TOOL_CALL]/
    23→// [TOOL_RESULT] history markers). The system prompt is mode-aware
    24→// (buildAgentSystemPrompt) and carries the active project's tree + commits.
    25→// Successful project tools emit WS project events to the user room; a turn
    26→// that wrote/deleted files in "act" mode gets one auto-checkpoint BEFORE the
    27→// final answer streams (nothing slow may run after message:end — the busy
    28→// flag clears as soon as the turn returns).
    29→//
    30→// Path MUST be "/" (Caddy gateway requirement), port 3003 (hardcoded).

   128→/**
   129→ * Mode + active-project system prompt for the turn (contract 3-ctr §4).
   130→ * Built ONCE per turn (not per iteration). Workspace failures degrade to a
   131→ * project-less prompt — a broken tree must never kill the turn.
   132→ */
   133→async function buildTurnSystemPrompt(
   134→  userId: string,
   135→  thread: ThreadTurnInfo,
   136→): Promise<string> {
   137→  try {
   138→    // Active plan (any mode — act works through it, plan replaces it).
   139→    const tasks = await threadTasks(thread.id);
   140→    const planTasks = tasks.map((t) => ({ text: t.text, done: t.done }));
   141→
   142→    if (!thread.projectId) {
   143→      return buildAgentSystemPrompt({ mode: thread.mode, planTasks });
   144→    }
   145→    const project = await db.project.findFirst({
   146→      where: { id: thread.projectId, userId },
   147→      select: { id: true, name: true, origin: true },
   148→    });
   149→    if (!project) {
   150→      return buildAgentSystemPrompt({ mode: thread.mode });
   151→    }
   152→    const root = projectRoot(project.id);
   153→    const [tree, commits] = await Promise.all([
   154→      listWorkspaceTree(root),
   155→      listProjectCommits(root, PROMPT_COMMITS),
   156→    ]);
   157→    return buildAgentSystemPrompt({
   158→      mode: thread.mode,
   159→      projectName: project.name,
   160→      projectOrigin: project.origin,
   161→      projectTree: tree.entries
   162→        .filter((e) => e.type === "file")
   163→        .map((e) => e.path)
   164→        .slice(0, PROMPT_TREE_PATHS),
   165→      recentCommits: commits.map((c) => `${c.short} ${c.message}`),
   166→      planTasks,
   167→    });
   168→  } catch (err) {
   169→    console.warn(
   170→      "[agent] project context build failed (continuing without it):",
   171→      err instanceof Error ? err.message : String(err),
   172→    );
   173→    return buildAgentSystemPrompt({ mode: thread.mode });
   174→  }
   175→}

   179→/**
   180→ * LLM history from the last HISTORY_LIMIT message rows of the thread:
   181→ *  - user/assistant rows → {role, content} as-is (empty assistant stubs skipped)
   182→ *  - tool rows → rendered in the SAME shape as the wire protocol so the
   183→ *    model never sees a second format it could mimic:
   184→ *      assistant: {"tool":"<name>","args":{...}}
   185→ *      user:      [TOOL_RESULT] <toolResult ≤2000 chars>
   186→ *    Tool rows still marked "pending" (crashed turn) are skipped entirely.
   187→ */
   188→async function buildLLMHistory(threadId: string): Promise<LlmMessage[]> {
   189→  const rows = await db.message.findMany({
   190→    where: { threadId },
   191→    orderBy: { createdAt: "asc" },
   192→    take: -HISTORY_LIMIT, // last 30 rows, still ascending
   193→  });
   194→
   195→  const history: LlmMessage[] = [];
   196→  for (const row of rows) {
   197→    if (row.role === "user" || row.role === "assistant") {
   198→      if (!row.content.trim()) continue;
   199→      history.push({ role: row.role, content: row.content });
   200→    } else if (row.role === "tool") {
   201→      if (!row.toolResult || row.toolResult === "pending") continue;
   202→      const result =
   203→        row.toolResult.length > TOOL_RESULT_HISTORY_CHARS
   204→          ? row.toolResult.slice(0, TOOL_RESULT_HISTORY_CHARS)
   205→          : row.toolResult;
   206→      // The call, exactly as the model itself would emit it (assistant role) …
   207→      history.push({
   208→        role: "assistant",
   209→        content: `{"tool":${JSON.stringify(row.toolName ?? "unknown")},"args":${row.toolArgs ?? "{}"}}`,
   210→      });
   211→      // … and the result as a user-role system-ish message.
   212→      history.push({
   213→        role: "user",
   214→        content: `[TOOL_RESULT] ${result}`,
   215→      });
   216→    }
   217→  }
   218→  return history;
   219→}

   221→/**
   222→ * Clean a would-be plain-text LLM answer: strip leaked protocol artifacts
   223→ * ([TOOL_CALL …] / [TOOL_RESULT …] lines, bare tool-JSON objects). Returns
   224→ * null when nothing human-readable remains (caller should keep looping).
   225→ */
   226→function sanitizeTextAnswer(raw: string): string | null {
   227→  let text = raw;
   228→  // 1. Remove whole-line protocol markers.
   229→  text = text
   230→    .split("\n")
   231→    .filter((line) => !/^\s*\[(TOOL_CALL|TOOL_RESULT)/i.test(line))
   232→    .join("\n");
   233→  // 2. Drop whole-line bare tool-JSON objects (chained-call leak: the model
   234→  //    sometimes packs several {"tool":…} objects into one reply).
   235→  text = text
   236→    .split("\n")
   237→    .filter((line) => !/^\s*\{\s*"tool"\s*:/.test(line))
   238→    .join("\n");
   239→  // 3. Remove a bare JSON tool object if it is the entire message.
   240→  const trimmed = text.trim();
   241→  if (/^\{\s*"tool"\s*:/.test(trimmed)) {
   242→    try {
   243→      const obj = JSON.parse(trimmed);
   244→      if (typeof obj?.tool === "string") return null; // pure tool JSON → not text
   245→    } catch {
   246→      // fall through — partial JSON mixed with prose, keep going
   247→    }
   248→  }
   249→  const cleaned = text.trim();
   250→  return cleaned.length > 0 ? cleaned : null;
   251→}

   253→/**
   254→ * Stream a final text answer to the thread room (emulated streaming):
   255→ * assistant row (empty) → message:start → message:delta ×N → persist →
   256→ * message:end.
   257→ */
   258→async function streamFinalResponse(
   259→  room: string,
   260→  threadId: string,
   261→  text: string,
   262→): Promise<void> {
   263→  let assistantMessage = await db.message.create({
   264→    data: { threadId, role: "assistant", content: "" },
   265→  });
   266→  io.to(room).emit("message:start", { threadId, messageId: assistantMessage.id });
   267→
   268→  const chunks = chunkText(text);
   269→  for (const delta of chunks) {
   270→    io.to(room).emit("message:delta", { threadId, messageId: assistantMessage.id, delta });
   271→    await sleep(25 + Math.random() * 10); // ~25–35ms per chunk
   272→  }
   273→
   274→  assistantMessage = await db.message.update({
   275→    where: { id: assistantMessage.id },
   276→    data: { content: text },
   277→  });
   278→  io.to(room).emit("message:end", {
   279→    threadId,
   280→    message: serializeMessage(assistantMessage),
   281→  });
   282→}

   326→/**
   327→ * Parse a ```план fence from a plan-mode final answer.
   328→ * Lenient line formats: "- [ ] text", "- [x] text", "* [ ] text", "1. text".
   329→ * Returns null when no fence / no parseable lines (the prose stays as-is).
   330→ */
   331→export function parsePlanFence(text: string): string[] | null {
   332→  const fence = text.match(/```(?:план|plan)\s*\n([\s\S]*?)```/i);
   333→  if (!fence) return null;
   334→  const lines = fence[1]
   335→    .split("\n")
   336→    .map((line) =>
   337→      line
   338→        .replace(/^\s*(?:[-*+]|\d+[.)])\s*(?:\[[ xX]\])?\s*/, "")
   339→        .trim(),
   340→    )
   341→    .filter((line) => line.length > 0)
   342→    .map((line) => line.slice(0, PLAN_MAX_TEXT_CHARS))
   343→    .slice(0, PLAN_MAX_TASKS);
   344→  return lines.length > 0 ? lines : null;
   345→}

   397→/**
   398→ * Agent turn (tool-calling loop):
   399→ *  1. persist user message → emit "message:user" → bump thread.updatedAt
   400→ *  2. build the mode/project system prompt ONCE (buildTurnSystemPrompt)
   401→ *  3. up to MAX_TOOL_ITERATIONS LLM iterations:
   402→ *       - "agent:thinking" (before every LLM call, incl. the first)
   403→ *       - history = last 30 rows (user/assistant + [TOOL_CALL]/[TOOL_RESULT])
   404→ *       - LLM reply parses as a tool call? → persist tool row ("pending") →
   405→ *         emit "tool:start" → execute with ToolContext → persist result
   406→ *         (≤8000 chars) → emit "tool:end" (+ project WS events on success) →
   407→ *         next iteration sees the result in history
   408→ *       - plain text? → remembered as finalText, loop ends
   409→ *  4. dirty act-mode turn (write_file/delete_file succeeded) →
   410→ *     auto-checkpoint (failures are logged and swallowed)
   411→ *  5. stream the final answer (or the fallback when the loop ended without
   412→ *     one) — AFTER the checkpoint, so nothing slow runs after message:end
   413→ *  6. loop exhausted / turn timeout (>120s) without a text answer →
   414→ *     fallback message
   415→ *  7. auto-title brand-new threads → "thread:updated" to the user room
   416→ * Never throws — failures emit "error" to the triggering socket.
   417→ */
   418→async function runAgentTurn(
   419→  socket: Socket,
   420→  user: WsUser,
   421→  threadId: string,
   422→  content: string,
   423→  thread: ThreadTurnInfo,
   424→): Promise<void> {
   425→  const room = `thread:${threadId}`;
   426→  const userRoom = `user:${user.sub}`;
   427→  const turnStart = Date.now();
   428→
   429→  try {
   430→    // 1. Persist user message.
   431→    const userMessage = await db.message.create({
   432→      data: { threadId, role: "user", content },
   433→    });
   434→    await db.thread.update({ where: { id: threadId }, data: { updatedAt: new Date() } });
   435→    io.to(room).emit("message:user", { message: serializeMessage(userMessage) });
   436→
   437→    // 1b. Auto-title fresh threads EARLY (before any answer streams) so
   438→    // that nothing slow remains after the final message:end emit — this
   439→    // keeps the busy-flag race window at effectively zero.
   440→    await maybeAutoTitle(threadId, user.sub);
   441→
   442→    // 2. Mode + project context for the whole turn (built once).
   443→    const systemPrompt = await buildTurnSystemPrompt(user.sub, thread);
   444→
   445→    // 3. Tool-calling loop.
   446→    let finalText: string | null = null;
   447→    let turnDirty = false; // write_file/delete_file succeeded this turn
   448→
   449→    for (let iteration = 0; iteration < MAX_TOOL_ITERATIONS; iteration++) {
   450→      if (Date.now() - turnStart > TURN_TIMEOUT_MS) break;
   451→
   452→      // Thinking indicator (before the first LLM call and between iterations).
   453→      io.to(room).emit("agent:thinking", { threadId });
   454→
   455→      const history = await buildLLMHistory(threadId);
   456→      const raw = await generateLLMResponse(systemPrompt, history);
   457→
   458→      const call = parseToolCall(raw);
   459→      if (!call) {
   460→        // Would-be plain text: strip leaked protocol artifacts first.
   461→        const clean = sanitizeTextAnswer(raw);
   462→        if (clean) {
   463→          finalText = clean;
   464→          break;
   465→        }
   466→        // Nothing human-readable left (model emitted protocol garbage) —
   467→        // keep looping; the model gets another chance to answer properly.
   468→        continue;
   469→      }
   470→
   471→      // Tool call → persist a "pending" tool row first (it becomes the
   472→      // message id reported to the client).
   473→      const toolMessage = await db.message.create({
   474→        data: {
   475→          threadId,
   476→          role: "tool",
   477→          content: "",
   478→          toolName: call.tool,
   479→          toolArgs: JSON.stringify(call.args),
   480→          toolResult: "pending",
   481→        },
   482→      });
   483→      io.to(room).emit("tool:start", {
   484→        threadId,
   485→        messageId: toolMessage.id,
   486→        tool: call.tool,
   487→        args: call.args,
   488→      });
   489→
   490→      // Execute (never throws into the loop — errors become tool results).
   491→      // ctx is rebuilt every iteration: create_project may have bound the
   492→      // thread's projectId in a previous iteration of this same turn.
   493→      const ctx: ToolContext = {
   494→        threadId,
   495→        mode: thread.mode,
   496→        projectId: thread.projectId,
   497→      };
   498→      let result: unknown;
   499→      try {
   500→        const tool = getTool(call.tool);
   501→        result = tool
   502→          ? await tool.execute(call.args, user.sub, ctx)
   503→          : { error: `Неизвестный инструмент: ${call.tool}` };
   504→      } catch (err) {
   505→        result = { error: err instanceof Error ? err.message : String(err) };
   506→      }
   507→
   508→      const resultJson = (JSON.stringify(result) ?? "{}").slice(0, MAX_TOOL_RESULT_CHARS);
   509→      await db.message.update({
   510→        where: { id: toolMessage.id },
   511→        data: { toolResult: resultJson },
   512→      });
   513→      io.to(room).emit("tool:end", {
   514→        threadId,
   515→        messageId: toolMessage.id,
   516→        tool: call.tool,
   517→        args: call.args,
   518→        result,
   519→      });
   520→
   521→      // Project WS events (contract 3-ctr §5) — emitted HERE by the
   522→      // transport layer, tools stay io-free.
   523→      const r = resultObject(result);
   524→      if (r && r.error === undefined) {
   525→        if (call.tool === "complete_task") {
   526→          // Live plan progress → push the fresh task list to the user room.
   527→          await emitTasksUpdated(threadId, user.sub);
   528→        } else if (call.tool === "create_project") {
   529→          const p = resultObject(r.project);
   530→          if (p && typeof p.id === "string") {
   531→            // Keep the local binding fresh for later iterations + the
   532→            // auto-checkpoint below.
   533→            thread.projectId = p.id;
   534→            io.to(userRoom).emit("project:created", {
   535→              project: { id: p.id, name: p.name, origin: p.origin },
   536→            });
   537→            await createNotification(
   538→              io,
   539→              user.sub,
   540→              "project_created",
   541→              `Агент создал проект «${p.name}»`,
   542→              "Проект создан из шаблона и привязан к диалогу",
   543→              p.id,
   544→            );
   545→          }
   546→        } else if (
   547→          (call.tool === "write_file" || call.tool === "delete_file") &&
   548→          thread.projectId
   549→        ) {
   550→          turnDirty = true;
   551→          io.to(userRoom).emit("project:updated", {
   552→            projectId: thread.projectId,
   553→            reason: "files",
   554→          });
   555→        } else if (
   556→          call.tool === "checkpoint" &&
   557→          thread.projectId &&
   558→          r.noop === false
   559→        ) {
   560→          io.to(userRoom).emit("project:updated", {
   561→            projectId: thread.projectId,
   562→            reason: "checkpoint",
   563→          });
   564→          const cp = resultObject(r.commit);
   565→          const commitMsg =
   566→            cp && typeof cp.message === "string" ? cp.message : "контрольная точка";
   567→          await notifyCheckpoint(user.sub, thread.projectId, commitMsg);
   568→        }
   569→      }
   570→      // Loop continues — the next iteration sees [TOOL_CALL]/[TOOL_RESULT].
   571→    }
   572→
   573→    // 4. Auto-checkpoint a dirty act-mode turn (contract 3-ctr §5) — BEFORE
   574→    // the final answer streams. Failure must never break the turn.
   575→    if (turnDirty && thread.mode === "act" && thread.projectId) {
   576→      try {
   577→        const cp = await checkpointProject(
   578→          projectRoot(thread.projectId),
   579→          AUTO_CHECKPOINT_MESSAGE,
   580→        );
   581→        if (!cp.noop) {
   582→          io.to(userRoom).emit("project:updated", {
   583→            projectId: thread.projectId,
   584→            reason: "checkpoint",
   585→          });
   586→          await notifyCheckpoint(user.sub, thread.projectId, cp.commit?.message ?? AUTO_CHECKPOINT_MESSAGE);
   587→        }
   588→      } catch (err) {
   589→        console.warn(
   590→          "[agent] auto-checkpoint failed (ignored):",
   591→          err instanceof Error ? err.message : String(err),
   592→        );
   593→      }
   594→    }
   595→
   596→    // 5. Final answer (or the fallback when the loop ended without text).
   597→    //    Plan-mode answers may carry a ```план fence → save as Task rows.
   598→    //    Done BEFORE streaming so the card appears with the answer.
   599→    const answerText = finalText ?? FALLBACK_REPLY;
   600→    if (thread.mode === "plan" && finalText) {
   601→      const plan = parsePlanFence(answerText);
   602→      if (plan) await savePlanTasks(threadId, user.sub, plan);
   603→    }
   604→    await streamFinalResponse(room, threadId, answerText);
   605→
   606→    // NOTE: nothing slow may happen after the final message:end emit —
   607→    // the busy flag clears right after this function returns, and a
   608→    // client that sends its next message on message:end would otherwise
   609→    // race into "Агент ещё отвечает…". Auto-title moved to the turn start,
   610→    // the auto-checkpoint runs before the answer streams.
   611→  } catch (err) {
   612→    console.error(
   613→      `[agent] turn failed (thread ${threadId}):`,
   614→      err instanceof Error ? err.message : String(err),
   615→    );
   616→    socket.emit("error", { message: "Не удалось получить ответ. Попробуйте ещё раз." });
   617→  }
   618→}

   620→/** If the thread is still untitled and this is one of its first 2 messages —
   621→ *  derive a short title from the first user message and notify the user room. */
   622→async function maybeAutoTitle(threadId: string, userId: string): Promise<void> {
   623→  const thread = await db.thread.findUnique({ where: { id: threadId } });
   624→  if (!thread || thread.userId !== userId || thread.title !== "Новый диалог") return;
   625→
   626→  const count = await db.message.count({ where: { threadId } });
   627→  if (count > 2) return;
   628→
   629→  const firstUser = await db.message.findFirst({
   630→    where: { threadId, role: "user" },
   631→    orderBy: { createdAt: "asc" },
   632→  });
   633→  if (!firstUser) return;
   634→
   635→  const title = deriveThreadTitle(firstUser.content);
   636→  if (!title) return;
   637→
   638→  const updated = await db.thread.update({ where: { id: threadId }, data: { title } });
   639→  io.to(`user:${userId}`).emit("thread:updated", {
   640→    thread: {
   641→      id: updated.id,
   642→      title: updated.title,
   643→      updatedAt: updated.updatedAt.toISOString(),
   644→    },
   645→  });
   646→}

   648→// ─────────────────────────── connection handling ───────────────────────────
   649→
   650→io.on("connection", async (socket: Socket) => {
   651→  // Auth: { token } in the handshake auth field.
   652→  const token = (socket.handshake.auth as { token?: unknown })?.token;
   653→  if (typeof token !== "string" || !token) {
   654→    socket.emit("error", { message: "Не авторизован" });
   655→    socket.disconnect(true);
   656→    return;
   657→  }
   658→
   659→  const user = await verifyWsToken(token);
   660→  if (!user) {
   661→    socket.emit("error", { message: "Не авторизован" });
   662→    socket.disconnect(true);
   663→    return;
   664→  }
   665→
   666→  socket.data.user = { userId: user.sub, email: user.email, name: user.name, role: user.role };
   667→  socket.join(`user:${user.sub}`);
   668→  console.log(`[ws] connected: ${user.email} (${socket.id})`);
   669→
   670→  // ── thread:join {threadId} ──
   671→  socket.on("thread:join", async (payload: unknown) => {
   672→    try {
   673→      const { threadId } = (payload ?? {}) as { threadId?: unknown };
   674→      if (typeof threadId !== "string" || !threadId) {
   675→        socket.emit("error", { message: "Некорректный запрос" });
   676→        return;
   677→      }
   678→      const thread = await db.thread.findUnique({ where: { id: threadId } });
   679→      if (!thread || thread.userId !== user.sub) {
   680→        socket.emit("error", { message: "Диалог не найден" });
   681→        return;
   682→      }
   683→      socket.join(`thread:${threadId}`);
   684→    } catch (err) {
   685→      console.error("[ws] thread:join failed:", err instanceof Error ? err.message : String(err));
   686→      socket.emit("error", { message: "Диалог не найден" });
   687→    }
   688→  });
   689→
   690→  // ── thread:leave {threadId} ──
   691→  socket.on("thread:leave", (payload: unknown) => {
   692→    const { threadId } = (payload ?? {}) as { threadId?: unknown };
   693→    if (typeof threadId !== "string" || !threadId) {
   694→      socket.emit("error", { message: "Некорректный запрос" });
   695→      return;
   696→    }
   697→    socket.leave(`thread:${threadId}`);
   698→  });
   699→
   700→  // ── message:send {threadId, content} ──
   701→  socket.on("message:send", async (payload: unknown) => {
   702→    try {
   703→      const { threadId, content } = (payload ?? {}) as { threadId?: unknown; content?: unknown };
   704→      if (typeof threadId !== "string" || !threadId || typeof content !== "string") {
   705→        socket.emit("error", { message: "Некорректный запрос" });
   706→        return;
   707→      }
   708→      const text = content.trim();
   709→      if (!text) {
   710→        socket.emit("error", { message: "Сообщение не может быть пустым" });
   711→        return;
   712→      }
   713→      if (text.length > MAX_CONTENT_LENGTH) {
   714→        socket.emit("error", { message: `Сообщение слишком длинное (максимум ${MAX_CONTENT_LENGTH} символов)` });
   715→        return;
   716→      }
   717→
   718→      const thread = await db.thread.findUnique({ where: { id: threadId } });
   719→      if (!thread || thread.userId !== user.sub) {
   720→        socket.emit("error", { message: "Диалог не найден" });
   721→        return;
   722→      }
   723→
   724→      // One agent turn per thread at a time.
   725→      if (runningThreads.has(threadId)) {
   726→        socket.emit("error", { message: "Агент ещё отвечает…" });
   727→        return;
   728→      }
   729→
   730→      runningThreads.set(threadId, true);
   731→      try {
   732→        // Make sure the sender receives thread room events even if they
   733→        // skipped an explicit thread:join.
   734→        socket.join(`thread:${threadId}`);
   735→        await runAgentTurn(socket, user, threadId, text, thread);
   736→      } finally {
   737→        runningThreads.delete(threadId);
   738→      }
   739→    } catch (err) {
   740→      console.error("[ws] message:send failed:", err instanceof Error ? err.message : String(err));
   741→      socket.emit("error", { message: "Не удалось отправить сообщение" });
   742→    }
   743→  });
   744→
   745→  socket.on("disconnect", () => {
   746→    console.log(`[ws] disconnected: ${user.email} (${socket.id})`);
   747→  });
   748→});

   763→  }
