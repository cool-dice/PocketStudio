    51→const TOKEN_STORAGE_KEY = "vf_token";
    52→
    53→export function setAuthToken(token: string): void {
    54→  try {
    55→    window.localStorage.setItem(TOKEN_STORAGE_KEY, token);
    56→  } catch {
    57→    // localStorage unavailable (private mode) — cookie fallback still applies.
    58→  }
    59→}
    60→
    61→export function clearAuthToken(): void {
    62→  try {
    63→    window.localStorage.removeItem(TOKEN_STORAGE_KEY);
    64→  } catch {
    65→    // ignore
    66→  }
    67→}
    68→
    69→function getAuthToken(): string | null {
    70→  try {
    71→    return window.localStorage.getItem(TOKEN_STORAGE_KEY);
    72→  } catch {
    73→    return null;
    74→  }
    75→}
    76→
    77→function authHeaders(): Record<string, string> {
    78→  const token = getAuthToken();
    79→  return token ? { authorization: `Bearer ${token}` } : {};
    80→}

    82→/** Blob → base64 (для загрузки бинарников в /upload). */
    83→function blobToBase64(blob: Blob): Promise<string> {
    84→  return new Promise((resolve, reject) => {
    85→    const reader = new FileReader();
    86→    reader.onerror = () => reject(new Error("Не удалось прочитать файл"));
    87→    reader.onload = () => {
    88→      const result = String(reader.result ?? "");
    89→      const comma = result.indexOf(",");
    90→      resolve(comma >= 0 ? result.slice(comma + 1) : result);
    91→    };
    92→    reader.readAsDataURL(blob);
    93→  });
    94→}

    96→export class ApiError extends Error {
    97→  status: number;
    98→  fields?: Record<string, string>;
    99→
   100→  constructor(message: string, status: number, fields?: Record<string, string>) {
   101→    super(message);
   102→    this.name = "ApiError";
   103→    this.status = status;
   104→    this.fields = fields;
   105→  }
   106→}
   107→
   108→async function request<T>(path: string, init?: RequestInit): Promise<T> {
   109→  let res: Response;
   110→  try {
   111→    res = await fetch(path, {
   112→      credentials: "same-origin",
   113→      ...init,
   114→      headers: {
   115→        "content-type": "application/json",
   116→        ...authHeaders(),
   117→        ...init?.headers,
   118→      },
   119→    });
   120→  } catch {
   121→    throw new ApiError("Нет соединения с сервером", 0);
   122→  }
   123→
   124→  const data: unknown = await res.json().catch(() => ({}));
   125→
   126→  if (!res.ok) {
   127→    // A stale/invalid Bearer token — drop it so the next login starts clean.
   128→    // 401 from login/register themselves must not wipe a valid session.
   129→    if (res.status === 401 && !/\/api\/auth\/(login|register)/.test(path)) {
   130→      clearAuthToken();
   131→    }
   132→    const body = (data ?? {}) as {
   133→      error?: string;
   134→      fields?: Record<string, string>;
   135→    };
   136→    throw new ApiError(
   137→      body.error ?? `Ошибка запроса (${res.status})`,
   138→      res.status,
   139→      body.fields,
   140→    );
   141→  }
   142→
   143→  return data as T;
   144→}

   146→/**
   147→ * Same as request() but for FormData bodies — the browser must set the
   148→ * multipart boundary itself, so NO content-type header is passed.
   149→ */
   150→async function requestForm<T>(path: string, form: FormData): Promise<T> {
   151→  let res: Response;
   152→  try {
   153→    res = await fetch(path, {
   154→      credentials: "same-origin",
   155→      method: "POST",
   156→      headers: authHeaders(),
   157→      body: form,
   158→    });
   159→  } catch {
   160→    throw new ApiError("Нет соединения с сервером", 0);
   161→  }
   162→
   163→  const data: unknown = await res.json().catch(() => ({}));
   164→
   165→  if (!res.ok) {
   166→    if (res.status === 401) clearAuthToken();
   167→    const body = (data ?? {}) as {
   168→      error?: string;
   169→      fields?: Record<string, string>;
   170→    };
   171→    throw new ApiError(
   172→      body.error ?? `Ошибка запроса (${res.status})`,
   173→      res.status,
   174→      body.fields,
   175→    );
   176→  }
   177→
   178→  return data as T;
   179→}

   616→    try {
   617→      res = await fetch(
   618→        `/api/workspaces/${encodeURIComponent(projectId)}/export`,
   619→        {
   620→          credentials: "same-origin",
   621→          headers: authHeaders(),
   622→        },
   623→      );
   624→    } catch {
   625→      throw new ApiError("Нет соединения с сервером", 0);
   626→    }
   627→    if (!res.ok) {
   628→      if (res.status === 401) clearAuthToken();
   629→      const body = (await res.json().catch(() => ({}))) as { error?: string };
   630→      throw new ApiError(
   631→        body.error ?? `Ошибка запроса (${res.status})`,
   632→        res.status,
   633→      );
   634→    }
   635→    return res.blob();
   636→  },
