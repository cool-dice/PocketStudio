     2→//
     3→// The z-ai SDK has no native function calling, so tools are described in the
     4→// system prompt (JSON protocol) and the model's JSON reply is parsed by
     5→// agent.ts:parseToolCall. Every tool is userId-scoped and validates its args
     6→// manually — invalid args return {error: "..."} instead of throwing, so the
     7→// tool-calling loop in server.ts never crashes.

     9→// Project/file tools (Stage 3, worklog Task 3-ctr) wrap the pure-node

    11→// turn-context aware: server.ts passes ToolContext {threadId, mode,
    12→// projectId} from the thread row. Write tools self-check the "act" mode.
    13→// WS events (project:created / project:updated) are emitted by server.ts,
    14→// NOT here — tools stay io-free.

    16→// Allowlists MUST stay in sync with the main app's notes REST contract

   100→/** First 140 chars of a note text as a single line. */
   101→function makePreview(text: string | null): string {
   102→  const single = (text ?? "").replace(/\s+/g, " ").trim();
   103→  return single.length <= 140 ? single : single.slice(0, 140);
   104→}
   105→
   106→/**
   107→ * limit arg: absent/invalid → default 10; out-of-range → clamped to 1..20.
   108→ * (Lenient on purpose: an odd limit from the LLM should not fail the turn.)
   109→ */
   110→function parseLimit(value: unknown, fallback = 10): number {
   111→  const n =
   112→    typeof value === "number"
   113→      ? value
   114→      : typeof value === "string" && value.trim() !== "" && !isNaN(Number(value))
   115→        ? Number(value)
   116→        : NaN;
   117→  if (!Number.isFinite(n)) return fallback;
   118→  return Math.min(20, Math.max(1, Math.floor(n)));
   119→}

   121→/** Find the user's category by exact name, case-insensitive (SQLite-aware). */
   122→async function findCategoryByName(
   123→  userId: string,
   124→  name: string,
   125→): Promise<{ id: string; name: string; color: string; icon: string } | null> {
   126→  const lower = name.toLowerCase();

   128→  return categories.find((c) => c.name.toLowerCase() === lower) ?? null;

   152→const createNote: ToolDef = {
   153→  name: "create_note",
   154→  description:
   155→    "Записать заметку пользователя в блокнот. Возвращает созданную заметку и её категорию.",
   156→  argsSchema: {
   157→    text: "полный текст заметки (обязательно, 1–5000 символов)",
   158→    category_name: "название категории (необязательно, 1–40 символов)",
   159→    category_color:
   160→      "цвет категории: emerald|amber|rose|sky|violet|stone|teal|orange|pink|cyan (по умолчанию stone)",
   161→    category_icon:
   162→      "иконка категории: lightbulb|briefcase|shopping-cart|heart|brain|zap|star|book|code|rocket|wallet|coffee (по умолчанию lightbulb)",
   163→  },
   164→  async execute(args: any, userId: string) {
   165→    if (typeof args !== "object" || args === null) {
   166→      return { error: "Некорректные аргументы инструмента" };
   167→    }
   168→
   169→    // text (required, 1..5000 after trim)
   170→    if (typeof args.text !== "string") {
   171→      return { error: "Аргумент text обязателен и должен быть строкой" };
   172→    }
   173→    const text = args.text.trim();
   174→    if (!text) return { error: "Текст заметки не может быть пустым" };
   175→    if (text.length > 5000) {
   176→      return { error: "Текст заметки слишком длинный (максимум 5000 символов)" };
   177→    }
   178→
   179→    // category_name (optional, 1..40 after trim; empty → treated as absent)
   180→    let categoryName: string | null = null;
   181→    if (args.category_name !== undefined && args.category_name !== null) {
   182→      if (typeof args.category_name !== "string") {
   183→        return { error: "Аргумент category_name должен быть строкой" };
   184→      }
   185→      const trimmed = args.category_name.trim();
   186→      if (trimmed.length > 40) {
   187→        return { error: "Название категории слишком длинное (максимум 40 символов)" };
   188→      }
   189→      categoryName = trimmed || null;
   190→    }
   191→
   192→    // category_color / category_icon — allowlist with default fallback
   193→    // (invalid value from the LLM degrades to the default, not an error)
   194→    const color =
   195→      typeof args.category_color === "string" &&
   196→      (CATEGORY_COLORS as readonly string[]).includes(args.category_color)
   197→        ? args.category_color
   198→        : DEFAULT_COLOR;
   199→    const icon =
   200→      typeof args.category_icon === "string" &&
   201→      (CATEGORY_ICONS as readonly string[]).includes(args.category_icon)
   202→        ? args.category_icon
   203→        : DEFAULT_ICON;
   204→
   205→    // Category: reuse the user's existing one (case-insensitive name match),
   206→    // create a new one only when a name was given and nothing matched.
   207→    let category = categoryName ? await findCategoryByName(userId, categoryName) : null;
   208→    let createdNewCategory = false;
   209→
   210→    if (categoryName && !category) {
   211→      try {
   212→        category = await db.category.create({
   213→          data: { userId, name: categoryName, color, icon },
   214→        });
   215→        createdNewCategory = true;
   216→      } catch (err) {
   217→        // Race on the unique [userId, name] constraint → re-fetch the winner.
   218→        const existing = await findCategoryByName(userId, categoryName);
   219→        if (existing) {
   220→          category = existing;
   221→        } else {
   222→          return {
   223→            error: "Не удалось создать категорию: " + (err instanceof Error ? err.message : String(err)),
   224→          };
   225→        }
   226→      }
   227→    }
   228→
   229→    const note = await db.note.create({
   230→      data: {
   231→        userId,
   232→        rawText: text,
   233→        status: "pending",
   234→        categoryId: category?.id ?? null,
   235→      },
   236→    });
   237→
   238→    return {
   239→      note: {
   240→        id: note.id,
   241→        rawText: note.rawText,
   242→        status: note.status,
   243→        createdAt: note.createdAt.toISOString(),
   244→      },
   245→      category: category ? categoryShape(category) : null,
   246→      createdNewCategory,
   247→    };
   248→  },
   249→};

   388→/** Normalize a subdir arg: strip "./" prefixes and slashes, keep hidden
   389→ *  names (".env") intact; rejects ".." segments. Returns "" for root. */
   390→function normalizeSubdir(raw: string): string {
   391→  let dir = raw.replace(/\\/g, "/").trim();
   392→  dir = dir.replace(/^(\.\/)+/, "").replace(/^\/+/, "").replace(/\/+$/, "");
   393→  if (dir === ".") return "";
   394→  return dir;
   395→}

   399→const createProject: ToolDef = {
   400→  name: "create_project",
   401→  description:
   402→    "Создать проект из шаблона Next.js (реальные файлы и git-репозиторий) и привязать его к текущему диалогу. Возвращает проект, число файлов и корневые файлы. Доступен только в режиме «Действовать».",
   403→  argsSchema: {
   404→    name: "название проекта (обязательно, 1–80 символов)",
   405→    description: "короткое описание проекта (необязательно, до 500 символов)",
   406→    note_id: "идентификатор заметки, которую связать с проектом (необязательно)",
   407→  },
   408→  async execute(args: any, userId: string, ctx: ToolContext) {
   409→    if (typeof args !== "object" || args === null) {
   410→      return { error: "Некорректные аргументы инструмента" };
   411→    }
   412→
   413→    // Mutation guard: creating projects materializes real files — only
   414→    // the «act» mode may do that (ask/plan/review must stay read-only).
   415→    if (ctx.mode !== "act") return { error: ACT_MODE_ERROR };
   416→
   417→    // name (required, 1..80 after trim)
   418→    if (typeof args.name !== "string") {
   419→      return { error: "Аргумент name обязателен и должен быть строкой" };
   420→    }
   421→    const name = args.name.trim();
   422→    if (!name) return { error: "Название проекта не может быть пустым" };
   423→    if (name.length > 80) {
   424→      return { error: "Название проекта слишком длинное (максимум 80 символов) }" };
   425→    }
   426→
   427→    // description (optional, ≤500 after trim)
   428→    let description: string | null = null;
   429→    if (args.description !== undefined && args.description !== null) {
   430→      if (typeof args.description !== "string") {
   431→        return { error: "Аргумент description должен быть строкой" };
   432→      }
   433→      const trimmed = args.description.trim();
   434→      if (trimmed.length > 500) {
   435→        return { error: "Описание слишком длинное (максимум 500 символов)" };
   436→      }
   437→      description = trimmed || null;
   438→    }
   439→
   440→    // note_id (optional string)
   441→    let noteId: string | null = null;
   442→    if (args.note_id !== undefined && args.note_id !== null) {
   443→      if (typeof args.note_id !== "string") {
   444→        return { error: "Аргумент note_id должен быть строкой" };
   445→      }
   446→      noteId = args.note_id.trim() || null;
   447→    }
   448→
   449→    // 1. DB row first (rootPath is filled once the workspace exists).
   450→    const project = await db.project.create({
   451→      data: { userId, name, description, origin: "template", rootPath: "" },
   452→    });
   453→
   454→    // 2. Materialize the workspace dir from the template + git init.
   455→    const root = projectRoot(project.id);
   456→    let tree: Awaited<ReturnType<typeof listWorkspaceTree>>;
   457→    try {
   458→      await createFromTemplate(root);
   459→      await initProjectGit(root);
   460→      tree = await listWorkspaceTree(root);
   461→    } catch (err) {
   462→      // Roll the row AND the directory back — no orphan state survives.
   463→      await removeProjectDir(project.id).catch(() => {});
   464→      await db.project.delete({ where: { id: project.id } }).catch(() => {});
   465→      return {
   466→        error:
   467→          "Не удалось создать проект: " +
   468→          (err instanceof Error ? err.message : String(err)),
   469→      };
   470→    }
   471→
   472→    // 3. Fill rootPath now that the workspace exists.
   473→    await db.project.update({ where: { id: project.id }, data: { rootPath: root } });
   474→
   475→    // 4. Bind the current thread — only when it has no project yet.
   476→    const thread = await db.thread.findUnique({ where: { id: ctx.threadId } });
   477→    if (thread && thread.userId === userId && thread.projectId === null) {
   478→      await db.thread.update({
   479→        where: { id: thread.id },
   480→        data: { projectId: project.id },
   481→      });
   482→    }
   483→
   484→    // 5. Optional originating note link (kind "proposal": the agent turned
   485→    //    the user's thought into a project). Unique race → already linked.
   486→    if (noteId) {
   487→      const note = await db.note.findFirst({ where: { id: noteId, userId } });
   488→      if (note) {
   489→        await db.noteLink
   490→          .create({ data: { noteId: note.id, projectId: project.id, kind: "proposal" } })
   491→          .catch(() => {});
   492→      }
   493→    }
   494→
   495→    return {
   496→      project: { id: project.id, name, origin: project.origin },
   497→      filesCount: tree.entries.filter((e) => e.type === "file").length,
   498→      rootFiles: tree.entries.filter((e) => !e.path.includes("/")).map((e) => e.path),
   499→    };
   500→  },
   501→};

   534→    path: "подкаталог внутри проекта (необязательно, по умолчанию — корень проекта)",
   535→  },
   536→  async execute(args: any, userId: string, ctx: ToolContext) {
   537→    if (typeof args !== "object" || args === null) {
   538→      return { error: "Некорректные аргументы инструмента" };
   539→    }
   540→
   541→    // path (optional subdir; prefix matching keeps it inside the project)
   542→    let subdir = "";
   543→    if (args.path !== undefined && args.path !== null) {
   544→      if (typeof args.path !== "string") {
   545→        return { error: "Аргумент path должен быть строкой" };
   546→      }
   547→      subdir = normalizeSubdir(args.path);
   548→    }
   549→    if (subdir.split("/").some((s) => s === "..")) {
   550→      return { error: "Путь вне проекта запрещён" };
   551→    }
   552→
   553→    const loaded = await loadProject(userId, ctx);
   554→    if ("error" in loaded) return { error: loaded.error };
   555→
   556→    try {
   557→      const root = projectRoot(loaded.project.id);
   558→      const tree = await listWorkspaceTree(root);
   559→
   560→      // Scope to the subdir and re-base the paths relative to it.
   561→      const prefix = subdir ? subdir + "/" : "";
   562→      const scoped = tree.entries
   563→        .filter((e) => (subdir ? e.path.startsWith(prefix) : true))
   564→        .map((e) => ({
   565→          path: subdir ? e.path.slice(prefix.length) : e.path,
   566→          type: e.type,
   567→          size: e.size,
   568→        }))
   569→        .filter((e) => e.path !== "");
   570→
   571→      return {
   572→        files: scoped.slice(0, MAX_LIST_FILES_ENTRIES),
   573→        truncated: tree.truncated || scoped.length > MAX_LIST_FILES_ENTRIES,
   574→      };
   575→    } catch (err) {
   576→      return { error: err instanceof Error ? err.message : String(err) };
   577→    }
   578→  },

   695→    if (args.message !== undefined && args.message !== null) {

   711→

   713→      const root = projectRoot(loaded.project.id);

   724→const completeTask: ToolDef = {
   725→  name: "complete_task",
   726→  description:
   727→    "Отметить шаг плана диалога выполненным. Вызывай сразу после того, как закончил работу по шагу (номер — позиция в плане, начиная с 1).",
   728→  argsSchema: {
   729→    task: "номер шага плана (целое число, 1-based)",
   730→  },
   731→  async execute(args: any, _userId: string, ctx: ToolContext) {
   732→    if (typeof args !== "object" || args === null) {
   733→      return { error: "Некорректные аргументы инструмента" };
   734→    }
   735→
   736→    // task number: 1-based position in the plan (lenient parse like parseLimit).
   737→    const n =
   738→      typeof args.task === "number"
   739→        ? args.task
   740→        : typeof args.task === "string" && args.task.trim() !== "" && !isNaN(Number(args.task))
   741→          ? Number(args.task)
   742→          : NaN;
   743→    if (!Number.isInteger(n) || n < 1 || n > 50) {
   744→      return { error: "Аргумент task должен быть целым номером шага (от 1)" };
   745→    }
   746→
   747→    const tasks = await db.task.findMany({
   748→      where: { threadId: ctx.threadId },
   749→      orderBy: { order: "asc" },
   750→    });
   751→    if (tasks.length === 0) {
   752→      return { error: "У диалога пока нет плана — сначала составьте план в режиме «План»" };
   753→    }
   754→    const target = tasks[n - 1] ?? null;
   755→    if (!target) {
   756→      return { error: `В плане ${tasks.length} шаг(ов) — шага №${n} нет` };
   757→    }
   758→    if (target.done) {
   759→      return { task: { id: target.id, order: target.order, text: target.text, done: true }, noop: true };
   760→    }
   761→
   762→    const updated = await db.task.update({
   763→      where: { id: target.id },
   764→      data: { done: true },
   765→    });
   766→    return {
   767→      task: { id: updated.id, order: updated.order, text: updated.text, done: updated.done },
   768→    };
   769→  },
   770→};
