    90→
    91→function IdeRegion({
    92→  el,
    93→  designer,
    94→  hovered,
    95→  selected,
    96→  onHover,
    97→  onSelect,
    98→  className,
    99→  badgeSide = "left",
   100→  dashedHover = false,
   101→  children,
   102→}: {
   103→  el: IdeElement;
   104→  designer: boolean;
   105→  hovered: boolean;
   106→  selected: boolean;
   107→  onHover: (id: string | null) => void;
   108→  onSelect: (id: string) => void;
   109→  className?: string;
   110→  badgeSide?: "left" | "right";
   111→  dashedHover?: boolean;
   112→  children: ReactNode;
   113→}) {
   114→  if (!designer) {
   115→    return <div className={className}>{children}</div>;
   116→  }
   117→  const outlined = hovered || selected;
   118→  return (
   119→    <div
   120→      role="button"
   121→      tabIndex={0}
   122→      aria-pressed={selected}
   123→      aria-label={`Элемент ${el.label}`}
   124→      className={cn(
   125→        "relative cursor-pointer outline-none",
   126→        className,
   127→        selected && "outline-2 outline-primary",
   128→        !selected &&
   129→          hovered &&
   130→          (dashedHover
   131→            ? "outline-1 outline-dashed outline-emerald-600/60"
   132→            : "outline-1 outline-primary/70"),
   133→      )}
   134→      onMouseEnter={() => onHover(el.id)}
   135→      onMouseLeave={() => onHover(null)}
   136→      onClick={(e) => {
   137→        e.stopPropagation();
   138→        onSelect(el.id);
   139→      }}
   140→      onKeyDown={(e) => {
   141→        if (e.key === "Enter" || e.key === " ") {
   142→          e.preventDefault();
   143→          e.stopPropagation();
   144→          onSelect(el.id);
   145→        }
   146→      }}
   147→    >
   148→      {outlined ? (
   149→        <span
   150→          className={cn(
   151→            "absolute -top-5 z-20 rounded bg-primary px-1.5 py-0.5 font-mono text-[10px] leading-4 whitespace-nowrap text-primary-foreground shadow-md",
   152→            badgeSide === "left" ? "left-0" : "right-0",
   153→          )}
   154→        >
   155→          {el.label}
   156→        </span>
   157→      ) : null}
   158→      {children}
   159→    </div>
   160→  );
   161→}
